cmake_minimum_required(VERSION 3.24)
project(hailstone_sentinel VERSION 1.0.0 LANGUAGES CXX)
if(APPLE)
  enable_language(OBJCXX)
  set(CMAKE_OBJCXX_STANDARD 20)
  set(CMAKE_OBJCXX_STANDARD_REQUIRED ON)
endif()
include(CTest)
find_package(Threads REQUIRED)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(HAILSTONE_SDK_DIR "${CMAKE_CURRENT_SOURCE_DIR}/sdk" CACHE PATH "Hailstone public SDK directory")
add_library(sentinel_core STATIC src/engine.cpp)
set_target_properties(sentinel_core PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_include_directories(sentinel_core PUBLIC src vendor)
add_library(sentinel_audio_data STATIC src/audio_data.cpp)
set_target_properties(sentinel_audio_data PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_include_directories(sentinel_audio_data PUBLIC src)
set(SENTINEL_PLATFORM_SOURCES)
set(SENTINEL_PLATFORM_LIBRARIES)
if(WIN32)
  list(APPEND SENTINEL_PLATFORM_LIBRARIES winmm sapi ole32)
else()
  enable_language(C)
  add_library(sentinel_miniaudio STATIC src/miniaudio_impl.c)
  target_include_directories(sentinel_miniaudio PRIVATE src vendor)
  set_target_properties(sentinel_miniaudio PROPERTIES POSITION_INDEPENDENT_CODE ON C_VISIBILITY_PRESET hidden)
  list(APPEND SENTINEL_PLATFORM_LIBRARIES sentinel_miniaudio ${CMAKE_DL_LIBS})
  if(APPLE)
    list(APPEND SENTINEL_PLATFORM_SOURCES src/speech_macos.mm)
    set_source_files_properties(src/speech_macos.mm PROPERTIES COMPILE_OPTIONS "-fobjc-arc")
    list(APPEND SENTINEL_PLATFORM_LIBRARIES "-framework Foundation" "-framework AVFoundation" "-framework CoreAudio" "-framework AudioToolbox" "-framework CoreFoundation")
  else()
    list(APPEND SENTINEL_PLATFORM_SOURCES src/speech_linux.cpp)
    list(APPEND SENTINEL_PLATFORM_LIBRARIES m)
  endif()
endif()
add_library(sentinel SHARED src/plugin.cpp src/audio.cpp ${SENTINEL_PLATFORM_SOURCES})
target_link_libraries(sentinel PRIVATE sentinel_core Threads::Threads)
target_link_libraries(sentinel PRIVATE sentinel_audio_data ${SENTINEL_PLATFORM_LIBRARIES})
target_include_directories(sentinel PRIVATE "${HAILSTONE_SDK_DIR}")
set_target_properties(sentinel PROPERTIES PREFIX "" CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN ON)
function(sentinel_warnings target)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /permissive- /EHsc /utf-8)
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
  endif()
endfunction()
sentinel_warnings(sentinel_core)
sentinel_warnings(sentinel_audio_data)
sentinel_warnings(sentinel)
add_custom_command(TARGET sentinel POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_directory
  "${CMAKE_CURRENT_SOURCE_DIR}/ui" "$<TARGET_FILE_DIR:sentinel>/sentinel/ui")
install(TARGETS sentinel LIBRARY DESTINATION sentinel RUNTIME DESTINATION sentinel)
install(DIRECTORY ui DESTINATION sentinel)
