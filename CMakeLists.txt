cmake_minimum_required(VERSION 3.24)
project(hailstone_sentinel VERSION 1.0.0 LANGUAGES CXX)
include(CTest)
find_package(Threads REQUIRED)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(HAILSTONE_SDK_DIR "${CMAKE_CURRENT_SOURCE_DIR}/sdk" CACHE PATH "Hailstone public SDK directory")
add_library(sentinel_core STATIC src/engine.cpp)
set_target_properties(sentinel_core PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_include_directories(sentinel_core PUBLIC src vendor)
add_library(sentinel SHARED src/plugin.cpp src/audio.cpp)
target_link_libraries(sentinel PRIVATE sentinel_core Threads::Threads)
target_include_directories(sentinel PRIVATE "${HAILSTONE_SDK_DIR}")
set_target_properties(sentinel PROPERTIES PREFIX "" CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN ON)
if(WIN32)
  target_link_libraries(sentinel PRIVATE winmm sapi ole32)
endif()
function(sentinel_warnings target)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /permissive- /EHsc /utf-8)
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
  endif()
endfunction()
sentinel_warnings(sentinel_core)
sentinel_warnings(sentinel)
add_custom_command(TARGET sentinel POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_directory
  "${CMAKE_CURRENT_SOURCE_DIR}/ui" "$<TARGET_FILE_DIR:sentinel>/sentinel/ui")
install(TARGETS sentinel LIBRARY DESTINATION sentinel RUNTIME DESTINATION sentinel)
install(DIRECTORY ui DESTINATION sentinel)
if(BUILD_TESTING)
  add_executable(sentinel_engine_tests tests/engine_test.cpp)
  target_link_libraries(sentinel_engine_tests PRIVATE sentinel_core)
  sentinel_warnings(sentinel_engine_tests)
  add_test(NAME sentinel_engine COMMAND sentinel_engine_tests)
  add_executable(sentinel_plugin_tests tests/plugin_test.cpp src/plugin.cpp src/audio.cpp)
  target_include_directories(sentinel_plugin_tests PRIVATE "${HAILSTONE_SDK_DIR}")
  target_compile_definitions(sentinel_plugin_tests PRIVATE SENTINEL_NO_AUDIO=1 SENTINEL_TESTING=1)
  target_link_libraries(sentinel_plugin_tests PRIVATE sentinel_core Threads::Threads)
  if(WIN32)
    target_link_libraries(sentinel_plugin_tests PRIVATE winmm sapi ole32)
  endif()
  sentinel_warnings(sentinel_plugin_tests)
  add_test(NAME sentinel_plugin COMMAND sentinel_plugin_tests)
endif()
